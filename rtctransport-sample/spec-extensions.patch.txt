This patch documents the C++ changes needed for the RtcTransport spec extensions.
Apply these changes to:
  third_party/blink/renderer/modules/peerconnection/rtc_transport/rtc_transport.cc

=== Change 1: Add OnSendOutcome to DatagramConnectionObserver ===
After the OnConnectionError() method, add:

  void OnSendOutcome(SendOutcome outcome) override {
    PostCrossThreadTask(
        *main_task_runner_, FROM_HERE,
        CrossThreadBindOnce(&RtcTransport::OnSendOutcomeOnMainThread,
                            MakeUnwrappingCrossThreadWeakHandle(transport_),
                            outcome));
  }

=== Change 2: Add OnSendOutcomeOnMainThread implementation ===
After the OnWritableChangeOnMainThread() method, add:

void RtcTransport::OnSendOutcomeOnMainThread(
    webrtc::DatagramConnection::Observer::SendOutcome outcome) {
  ExecutionContext* context = GetExecutionContext();
  if (!context) {
    return;
  }

  auto* sent_info = MakeGarbageCollected<RtcPacketSentInfo>();
  sent_info->setId(outcome.id);
  sent_info->setSendTime(
      outcome.send_time.IsFinite()
          ? RTCTimeStampFromTimeTicks(context,
                                     ConvertToBaseTimeTicks(outcome.send_time))
          : 0);
  sent_info->setBytesSent(outcome.bytes_sent);
  sent_info->setSuccess(outcome.status ==
                        webrtc::DatagramConnection::Observer::SendOutcome::
                            Status::kSuccess);
  sent_info_.push_back(sent_info);

  DispatchEvent(*Event::Create(event_type_names::kPendingpacketssentinfo));
}

=== Change 3: Add getPacketSentInfo implementation ===
After the getReceivedPackets() method, add:

HeapVector<Member<RtcPacketSentInfo>> RtcTransport::getPacketSentInfo() {
  HeapVector<Member<RtcPacketSentInfo>> info;
  std::swap(info, sent_info_);
  return info;
}

In OnPacketReceivedOnMainThread(), after pushing to received_packets_, add:

  DispatchEvent(*Event::Create(event_type_names::kPendingpacketsreceived));
